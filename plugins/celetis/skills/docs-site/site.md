# The site files

Only the skill `docs-site` reads this file. Each `##` heading after "The values" is the path of
one file of the site. Write the file with the text of its fence, filled with the values, and with
one newline at its end.

## The values

Replace each placeholder of this table with its value, and change no other text. The text
`${{ … }}` of the workflow is GitHub Actions syntax, not a placeholder. Before a value goes into a
file, put a `\` before each `\` and each `'` of the value.

| Placeholder | Value |
|---|---|
| `{{title}}` | The text of the first `#` heading of the root `README.md`; with no such heading, the name of the repository |
| `{{owner}}` | `owner.login` of `gh repo view` |
| `{{repo}}` | `name` of `gh repo view` |
| `{{branch}}` | `defaultBranchRef.name` of `gh repo view` |
| `{{bun}}` | The output of `bun --version` |
| `{{homes}}` | One line `  { folder: '<folder>', label: '<label>' },` for each published home, with two spaces before `{`. `<folder>` is the path of the home under `docs/`, with no `docs/` and no `/` at the end. The order is explanation, tutorials, how-to guides, reference, then the named homes. The labels are `Explanation`, `Tutorials`, `How-to guides`, and `Reference`, also for an existing home that takes the place of a default home; a named home has its folder as its label |

## website/homes.ts

````ts
import { existsSync, readdirSync } from 'node:fs';
import { join, resolve, sep } from 'node:path';

export type Home = { folder: string; label: string };

export const homes: Home[] = [
{{homes}}
];

const docs = resolve('../docs');

const documents = (folder: string): string[] =>
  existsSync(join(docs, folder))
    ? readdirSync(join(docs, folder), { recursive: true, encoding: 'utf8' })
        // The docs plugin globs with `exclude: []` and skips dot paths, so both count the same files.
        .filter((path) => path.endsWith('.md') && !path.split(sep).some((part) => part.startsWith('.')))
        .map((path) => join(folder, path))
        .sort()
    : [];

export const published: Home[] = homes.filter(({ folder }) => documents(folder).length > 0);

const first = homes[0];
export const homePage: string | undefined =
  first && existsSync(join(docs, first.folder, 'overview.md'))
    ? join(first.folder, 'overview.md')
    : published.map(({ folder }) => documents(folder)[0])[0];
````

## website/docusaurus.config.ts

````ts
import { existsSync, statSync } from 'node:fs';
import { dirname, isAbsolute, relative, resolve, sep } from 'node:path';
import { homePage, homes } from './homes.ts';

const repository = 'https://github.com/{{owner}}/{{repo}}';
const branch = '{{branch}}';
const root = resolve('..');

function githubUrl(source: string, url: string): string {
  const cut = url.indexOf('#');
  const path = cut < 0 ? url : url.slice(0, cut);
  const fragment = cut < 0 ? '' : url.slice(cut);
  const target = resolve(dirname(resolve(source)), decodeURIComponent(path));
  const inside = relative(root, target);
  if (inside === '..' || inside.startsWith(`..${sep}`) || isAbsolute(inside))
    throw new Error(`${source}: the link ${url} leaves the repository`);
  if (!existsSync(target)) throw new Error(`${source}: the link ${url} names no file`);
  const kind = statSync(target).isDirectory() ? 'tree' : 'blob';
  return `${repository}/${kind}/${branch}/${encodeURI(inside.split(sep).join('/'))}${fragment}`;
}

type Node = { type: string; url?: string; children?: Node[] };

const fileLinks = () => (tree: Node, file: { path: string }) => {
  const visit = (node: Node) => {
    if (
      (node.type === 'link' || node.type === 'definition') &&
      node.url !== undefined &&
      !/^[a-z][a-z\d+.-]*:/i.test(node.url) &&
      !node.url.startsWith('#') &&
      !node.url.startsWith('/') &&
      !/\.mdx?$/.test(node.url.split('#')[0] ?? '')
    )
      node.url = githubUrl(file.path, node.url);
    for (const child of node.children ?? []) visit(child);
  };
  visit(tree);
};

type FrontMatter = { frontMatter: Record<string, unknown>; content: string };
type FrontMatterParams = {
  filePath: string;
  fileContent: string;
  defaultParseFrontMatter: (params: { filePath: string; fileContent: string }) => Promise<FrontMatter>;
};

export default {
  title: '{{title}}',
  url: 'https://{{owner}}.github.io',
  baseUrl: '/{{repo}}/',
  trailingSlash: false,
  onBrokenLinks: 'throw',
  markdown: {
    format: 'detect',
    mermaid: true,
    parseFrontMatter: async (params: FrontMatterParams): Promise<FrontMatter> => {
      const result = await params.defaultParseFrontMatter(params);
      if (homePage === undefined || resolve(params.filePath) !== resolve('../docs', homePage)) return result;
      return { ...result, frontMatter: { ...result.frontMatter, slug: '/', sidebar_position: 0 } };
    },
    hooks: {
      onBrokenMarkdownLinks: ({ sourceFilePath, url }: { sourceFilePath: string; url: string }) =>
        githubUrl(sourceFilePath, url),
    },
  },
  themes: ['@docusaurus/theme-mermaid'],
  presets: [
    [
      'classic',
      {
        docs: {
          path: '../docs',
          include: homes.map(({ folder }) => `${folder}/**/*.md`),
          exclude: [],
          routeBasePath: '/',
          sidebarPath: './sidebars.ts',
          beforeDefaultRemarkPlugins: [fileLinks],
        },
        blog: false,
      },
    ],
  ],
  themeConfig: {
    navbar: {
      title: '{{title}}',
      items: [{ href: repository, label: 'GitHub', position: 'right' }],
    },
    // A wide diagram scales down to the page column, so the default font renders too small.
    mermaid: { options: { themeVariables: { fontSize: '1.5rem' } } },
  },
};
````

## .github/workflows/docs.yml

````yaml
name: docs
on:
  push:
    branches: [{{branch}}]
  workflow_dispatch:
concurrency:
  group: pages
  cancel-in-progress: false
jobs:
  build:
    runs-on: ubuntu-24.04
    permissions:
      contents: read
    steps:
      - uses: actions/checkout@v7
      - uses: oven-sh/setup-bun@v2
        with:
          bun-version: {{bun}}
      - name: Install site packages
        working-directory: website
        run: bun install --frozen-lockfile
      - name: Build site
        working-directory: website
        run: bun run --bun build
      - uses: actions/upload-pages-artifact@v5
        with:
          path: website/build
  deploy:
    needs: build
    runs-on: ubuntu-24.04
    permissions:
      pages: write
      id-token: write
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v5
````
